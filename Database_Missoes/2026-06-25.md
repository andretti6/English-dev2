// 1. Configurações do seu ambiente
const firebaseURL = "https://english-meetings-f08e4-default-rtdb.firebaseio.com/bhagDisplay.json";
const missaoNome = "The Hotel Nightmare"; // Digite o nome do arquivo .md da missão aqui

// 2. Buscar o arquivo na sua pasta do Obsidian
const arquivoMissao = app.vault.getAbstractFileByPath(`Database_Missoes/${missaoNome}.md`);

if (arquivoMissao) {
    // Ler o frontmatter (propriedades) do arquivo
    app.fileManager.processFrontMatter(arquivoMissao, async (frontmatter) => {
        const textoMissao = frontmatter.bhag || "No mission text found";
        
        // 3. Enviar via API (PUT) direto para o seu Firebase
        try {
            const response = await fetch(firebaseURL, {
                method: "PUT",
                body: JSON.stringify(textoMissao)
            });
            
            if(response.ok) {
                dv.paragraph(`✅ **Live Arena Updated!** Mission pushed to Firebase: *"${textoMissao}"*`);
            } else {
                dv.paragraph("❌ Error updating Firebase.");
            }
        } catch (error) {
            dv.paragraph("❌ Connection error: " + error);
        }
    });
} else {
    dv.paragraph(`⚠️ Mission file not found at: \`Database_Missoes/${missaoNome}.md\``);
}
